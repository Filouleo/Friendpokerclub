  <script>
    let appData = {};

    fetch('data_friends_poker_club_full.json')
      .then(res => res.json())
      .catch(() => fetch('data_friends_poker_club.json').then(res => res.json()))
      .then(data => {
        appData = data;
        initApp();
      });

    function initApp() {
      const membres = appData.membres || appData.members || [];
      const saisons = appData.saisons_historiques || appData.historical_seasons || [];

      document.getElementById('stats-counter').innerHTML = 
        `<span class="text-amber-400 font-bold">${membres.length}</span> Membres | <span class="text-amber-400 font-bold">${saisons.length || 14}</span> Saisons`;

      renderMembers(membres);
      renderFormats(appData.formats_de_jeu || []);
      renderReglements(appData.reglements_et_ethique || {});
      renderSaisons(saisons);
      renderDocs(appData.documents_source_archives || []);
    }

    function switchTab(tabId) {
      document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
      document.querySelectorAll('.nav-btn').forEach(btn => {
        btn.classList.remove('text-amber-400', 'bg-slate-800/80');
        btn.classList.add('text-slate-400');
      });
      document.getElementById(tabId).classList.remove('hidden');
      const activeBtn = document.getElementById('btn-' + tabId);
      if (activeBtn) {
        activeBtn.classList.add('text-amber-400', 'bg-slate-800/80');
        activeBtn.classList.remove('text-slate-400');
      }
    }

    function getLocalPdfPath(rawPath) {
      if (!rawPath || rawPath === '-') return null;
      const filename = rawPath.split('/').pop();
      return `pdf/${filename}`;
    }

    function renderMembers(members) {
      const grid = document.getElementById('members-grid');
      if (!grid) return;
      grid.innerHTML = members.map(m => {
        const pseudo = m.pseudonyme || m.pseudo;
        const nom = m.nom_prenom_reel || m.nom_real;
        const localPdf = getLocalPdfPath(m.lien_fiche_pdf || m.pdf_url);
        return `
          <div class="bg-slate-900 border border-slate-800 rounded-xl p-4 flex flex-col justify-between">
            <div>
              <h4 class="font-bold text-slate-100">${pseudo}</h4>
              <p class="text-xs text-slate-400 mb-3">${nom || 'Nom confidentiel'}</p>
            </div>
            ${localPdf 
              ? `<a href="${localPdf}" target="_blank" class="py-1.5 px-3 bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold text-xs rounded-lg text-center transition">
                  📄 Fiche PDF
                 </a>` 
              : `<span class="text-xs text-slate-600 italic">Sans fiche PDF</span>`}
          </div>`;
      }).join('');
    }

    document.getElementById('search-input')?.addEventListener('input', (e) => {
      const query = e.target.value.toLowerCase().trim();
      const allM = appData.membres || appData.members || [];
      const filtered = allM.filter(m => (m.pseudonyme || m.pseudo || '').toLowerCase().includes(query) || (m.nom_prenom_reel || m.nom_real || '').toLowerCase().includes(query));
      renderMembers(filtered);
    });

    function renderFormats(formats) {
      const container = document.getElementById('formats-container');
      if (container && formats.length) {
        container.innerHTML = formats.map(f => `
          <div class="bg-slate-900 border border-slate-800 rounded-xl p-5">
            <h3 class="font-bold text-amber-400 text-base mb-1">${f.nom}</h3>
            <p class="text-xs text-slate-300 font-semibold mb-2">${f.type}</p>
            <p class="text-xs text-slate-400">${f.structure}</p>
          </div>`).join('');
      }
    }

    function renderReglements(reg) {
      const charteList = document.getElementById('charte-list');
      if (charteList && reg.charte_ethique_gentleman) {
        charteList.innerHTML = reg.charte_ethique_gentleman.map(item => `<li class="flex items-start gap-2"><span class="text-amber-400">✔</span><span>${item}</span></li>`).join('');
      }
    }

    function renderSaisons(saisons) {
      const grid = document.getElementById('seasons-grid');
      if (grid && saisons.length) {
        grid.innerHTML = saisons.map(s => `
          <div class="bg-slate-900 border border-slate-800 rounded-xl p-4">
            <span class="text-xs font-bold text-amber-400">Saison ${s.saison || s.title}</span>
            <h4 class="font-bold text-slate-200 text-sm mt-1">${s.libelle || s.description}</h4>
          </div>`).join('');
      }
    }

    function renderDocs(docs) {
      const list = document.getElementById('docs-list');
      if (list && docs.length) {
        list.innerHTML = docs.map(d => `
          <div class="bg-slate-950 border border-slate-800 rounded-lg p-3 flex justify-between items-center">
            <span class="text-xs text-slate-300 font-medium">${d.titre}</span>
            <a href="pdf/${encodeURIComponent(d.fichier)}" target="_blank" class="text-xs bg-amber-500 text-slate-950 px-2.5 py-1 rounded font-bold">PDF</a>
          </div>`).join('');
      }
    }
  </script>
